cmake_minimum_required(VERSION 3.24)
project(FafSfdDiagnostics LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "The player diagnostics require Windows DirectSound")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(SFD_PLATFORM x64)
else()
  set(SFD_PLATFORM Win32)
endif()
get_filename_component(FAF_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(FafFfmpegRoot "${FAF_ROOT}/dependencies/ffmpeg/${SFD_PLATFORM}" CACHE PATH "FFmpeg development package")
set(FAF_SFD_ASSET_DIR "" CACHE PATH "Optional installed game movies directory")
option(FAF_SFD_ASAN "Instrument new backend and tests with AddressSanitizer" OFF)
set(SFD_ROOT "${FAF_ROOT}/src/sdk/moho/movie/sfd")
add_library(faf-sfd STATIC
  "${SFD_ROOT}/SfdHeaderParser.cpp" "${SFD_ROOT}/SfdMetadata.cpp"
  "${SFD_ROOT}/FfmpegSfdDecoder.cpp" "${SFD_ROOT}/DirectSoundMovieSink.cpp"
  "${SFD_ROOT}/IndependentSfdBackend.cpp"
  "${FAF_ROOT}/src/sdk/moho/audio/SofdecRuntime.cpp")
target_include_directories(faf-sfd PUBLIC "${FAF_ROOT}/src/sdk")
target_include_directories(faf-sfd SYSTEM PRIVATE "${FafFfmpegRoot}/include")
foreach(library avformat avcodec avutil swscale swresample)
  target_link_libraries(faf-sfd PUBLIC "${FafFfmpegRoot}/lib/${library}.lib")
endforeach()
target_link_libraries(faf-sfd PUBLIC dsound dxguid user32)
add_executable(sfd-inspect main.cpp)
target_link_libraries(sfd-inspect PRIVATE faf-sfd)
add_executable(sfd-parser-tests parser-tests.cpp "${SFD_ROOT}/SfdHeaderParser.cpp")
target_include_directories(sfd-parser-tests PRIVATE "${FAF_ROOT}/src/sdk")
foreach(target faf-sfd sfd-inspect sfd-parser-tests)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  if(FAF_SFD_ASAN)
    # MSVC ASan is incompatible with /RTC1 and incremental linking.
    target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
    target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
  endif()
endforeach()
file(GLOB FFMPEG_DLLS "${FafFfmpegRoot}/bin/*.dll")
foreach(dll IN LISTS FFMPEG_DLLS)
  add_custom_command(TARGET sfd-inspect POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${dll}" "$<TARGET_FILE_DIR:sfd-inspect>")
endforeach()
add_test(NAME parser COMMAND sfd-parser-tests)
if(FAF_SFD_ASSET_DIR)
  add_test(NAME portrait-decode COMMAND sfd-inspect --decode "${FAF_SFD_ASSET_DIR}/Abasi.sfd")
  add_test(NAME embedded-audio-decode COMMAND sfd-inspect --decode "${FAF_SFD_ASSET_DIR}/e3_demo_cut.sfd")
  add_test(NAME portrait-playback COMMAND sfd-inspect --play "${FAF_SFD_ASSET_DIR}/Abasi.sfd")
  add_test(NAME embedded-audio-playback COMMAND sfd-inspect --play-sound "${FAF_SFD_ASSET_DIR}/e3_demo_cut.sfd")
  add_test(NAME embedded-audio-nosound COMMAND sfd-inspect --play "${FAF_SFD_ASSET_DIR}/e3_demo_cut.sfd")
  add_test(NAME special-composition-rejected COMMAND sfd-inspect --reject "${FAF_SFD_ASSET_DIR}/FMV_loading02.sfd")
  set_tests_properties(portrait-playback embedded-audio-playback embedded-audio-nosound PROPERTIES TIMEOUT 45)
endif()
