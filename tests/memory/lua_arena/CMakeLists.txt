cmake_minimum_required(VERSION 3.24)
project(LuaArenaHostTests LANGUAGES CXX)
# Match the engine and GoogleTest static runtime on MSVC.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Lua arena host tests require x64")
endif()
set(FAF_LUA_ARENA_SANITIZER "none" CACHE STRING "none, address, undefined, or thread")
if(FAF_LUA_ARENA_SANITIZER STREQUAL "address")
  add_compile_options(-g -fsanitize=address -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address)
elseif(FAF_LUA_ARENA_SANITIZER STREQUAL "undefined")
  add_compile_options(-g -fsanitize=undefined -fno-sanitize-recover=all)
  add_link_options(-fsanitize=undefined)
elseif(FAF_LUA_ARENA_SANITIZER STREQUAL "thread")
  add_compile_options(-g -fsanitize=thread)
  add_link_options(-fsanitize=thread)
elseif(NOT FAF_LUA_ARENA_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown sanitizer")
endif()
include(../LuaArenaTests.cmake)
