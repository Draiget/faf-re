cmake_minimum_required(VERSION 3.25)
project(ParticleAllocationHostTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(FAF_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(FAF_GTEST_DIR "${FAF_SOURCE_ROOT}/dependencies/googletest" CACHE PATH "Official GoogleTest source")
set(FAF_GTEST_LIBRARY_DIR "" CACHE PATH "Optional matching prebuilt GoogleTest library directory")
option(FAF_PARTICLE_SANITIZE "Enable ASan and UBSan" OFF)
if(FAF_GTEST_LIBRARY_DIR)
  foreach(name gtest gtest_main)
    add_library(${name} STATIC IMPORTED)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${FAF_GTEST_LIBRARY_DIR}/lib${name}.a" INTERFACE_INCLUDE_DIRECTORIES "${FAF_GTEST_DIR}/googletest/include")
  endforeach()
else()
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  add_subdirectory("${FAF_GTEST_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
endif()
set(wm_root "${FAF_SOURCE_ROOT}/dependencies/WildMagic3p8/Foundation")
file(GLOB wm_subdirs LIST_DIRECTORIES true "${wm_root}/*")
set(wm_includes "${wm_root}")
foreach(path IN LISTS wm_subdirs)
  if(IS_DIRECTORY "${path}")
    list(APPEND wm_includes "${path}")
  endif()
endforeach()
file(GLOB_RECURSE source_inputs CONFIGURE_DEPENDS "${FAF_SOURCE_ROOT}/src/moho/particles/*.h" "${FAF_SOURCE_ROOT}/src/moho/particles/*.cpp" "${FAF_SOURCE_ROOT}/src/legacy/containers/*.h" "${FAF_SOURCE_ROOT}/src/legacy/containers/*.cpp")
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${generated}/ParticleBatchProduction.inc" "${generated}/ParticlePayload.inc" "${generated}/ParticlePeek.inc"
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/bind_production.py" --source-root "${FAF_SOURCE_ROOT}" --output "${generated}/ParticleBatchProduction.inc" --inject-append
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/bind_payload.py" --source-root "${FAF_SOURCE_ROOT}" --output "${generated}/ParticlePayload.inc"
 DEPENDS bind_production.py bind_payload.py ${source_inputs} VERBATIM)
add_custom_target(particle_allocation_bindings DEPENDS "${generated}/ParticleBatchProduction.inc" "${generated}/ParticlePayload.inc" "${generated}/ParticlePeek.inc")
foreach(target particle_allocation_host particle_allocation_faults)
 add_executable(${target} ParticleUploadTests.cpp)
 add_dependencies(${target} particle_allocation_bindings)
 target_compile_definitions(${target} PRIVATE FAF_RETAIL_LAYOUT=0 APPEND_HOOK)
 target_include_directories(${target} PRIVATE "${FAF_SOURCE_ROOT}/src" "${generated}" ${wm_includes})
 target_link_libraries(${target} PRIVATE gtest_main gtest Threads::Threads)
 set_target_properties(${target} PROPERTIES FOLDER "Tests/Particles")
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target} PRIVATE -O2 -g)
  if(FAF_PARTICLE_SANITIZE)
   target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer -fno-pie)
   target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -no-pie)
  endif()
 endif()
endforeach()
target_compile_definitions(particle_allocation_faults PRIVATE PARTICLE_ALLOCATION_TEST)
enable_testing()
foreach(batch 0 1)
 add_test(NAME ParticleAllocation_Batch_${batch} COMMAND particle_allocation_host)
 set_tests_properties(ParticleAllocation_Batch_${batch} PROPERTIES LABELS "Particles;ParticlesHost" ENVIRONMENT "PARTICLE_BATCH=${batch}")
endforeach()
add_test(NAME ParticleAllocation_Ownership COMMAND particle_allocation_faults --gtest_filter=*AllocationFailure*:*ActiveVectorFailure*:*RealInitializer*:*ExhaustionRemains*:*LinkedWorkItemOwns*:*AllocationCallback*)
set_tests_properties(ParticleAllocation_Ownership PROPERTIES LABELS "Particles;ParticlesHost;ParticlesAllocation")
