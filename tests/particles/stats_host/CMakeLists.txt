cmake_minimum_required(VERSION 3.25)
project(ParticleStatsHostTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
get_filename_component(FAF_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(FAF_GTEST_DIR "${FAF_SOURCE_ROOT}/dependencies/googletest" CACHE PATH "Existing official GoogleTest source")
set(FAF_GTEST_LIBRARY_DIR "" CACHE PATH "Optional matching prebuilt GTest library directory")
option(FAF_PARTICLE_SANITIZE "Enable ASan and UBSan" OFF)
set(FAF_STATS_NEGATIVE_CONTROL "" CACHE STRING "Mutation in generated test copies only")
if(FAF_GTEST_LIBRARY_DIR)
  foreach(name gtest gtest_main)
    add_library(${name} STATIC IMPORTED)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${FAF_GTEST_LIBRARY_DIR}/lib${name}.a" INTERFACE_INCLUDE_DIRECTORIES "${FAF_GTEST_DIR}/googletest/include")
  endforeach()
else()
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  add_subdirectory("${FAF_GTEST_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
endif()
set(wm_root "${FAF_SOURCE_ROOT}/dependencies/WildMagic3p8/Foundation")
file(GLOB wm_subdirs LIST_DIRECTORIES true "${wm_root}/*")
set(wm_includes "${wm_root}")
foreach(path IN LISTS wm_subdirs)
  if(IS_DIRECTORY "${path}")
    list(APPEND wm_includes "${path}")
  endif()
endforeach()
file(GLOB_RECURSE inputs CONFIGURE_DEPENDS "${FAF_SOURCE_ROOT}/src/moho/particles/*.h" "${FAF_SOURCE_ROOT}/src/moho/particles/*.cpp" "${FAF_SOURCE_ROOT}/src/legacy/containers/*.h" "${FAF_SOURCE_ROOT}/src/legacy/containers/*.cpp" "${wm_root}/*.h" "${wm_root}/*.inl" "${wm_root}/*.mcr")
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${generated}/Payload.inc" "${generated}/Production.inc" "${generated}/Lifecycle.inc" "${generated}/SourceContracts.inc" "${generated}/manifest.json"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/bind_stats.py" --source-root "${FAF_SOURCE_ROOT}" --output-dir "${generated}" "--negative-control=${FAF_STATS_NEGATIVE_CONTROL}"
  DEPENDS bind_stats.py ${inputs}
    "${FAF_SOURCE_ROOT}/src/moho/sim/WldSessionControl.cpp"
    "${FAF_SOURCE_ROOT}/src/moho/sim/CWldSessionCore.cpp"
    "${FAF_SOURCE_ROOT}/src/moho/render/SParticleBuffer.cpp"
    "${FAF_SOURCE_ROOT}/src/moho/render/d3d/CD3DDevice.cpp"
  VERBATIM)
add_executable(particle_stats_host StatsHostTests.cpp "${generated}/Production.inc" "${generated}/Payload.inc" "${generated}/Lifecycle.inc" "${generated}/SourceContracts.inc")
target_compile_definitions(particle_stats_host PRIVATE FAF_RETAIL_LAYOUT=0)
target_include_directories(particle_stats_host PRIVATE "${FAF_SOURCE_ROOT}/src" "${generated}" ${wm_includes})
target_link_libraries(particle_stats_host PRIVATE gtest_main gtest Threads::Threads)
set_target_properties(particle_stats_host PROPERTIES FOLDER "Tests/Particles")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(particle_stats_host PRIVATE -O2 -g)
  if(FAF_PARTICLE_SANITIZE)
    target_compile_options(particle_stats_host PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer -fno-pie)
    target_link_options(particle_stats_host PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -no-pie)
  endif()
elseif(FAF_PARTICLE_SANITIZE)
  message(FATAL_ERROR "This standalone sanitizer configuration requires GCC or Clang")
endif()
enable_testing()
add_test(NAME ParticleStats_Host COMMAND particle_stats_host)
set_tests_properties(ParticleStats_Host PROPERTIES LABELS "Particles;ParticlesHost;ParticleStats" ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
